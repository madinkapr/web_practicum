function isSpecial(n){
        let sum = 0;
        while(n>0){
            let last = n%10;
            sum+=last;
            n=Math.floor(n/10);
        }
        if(n%(sum**2)==0){
            console.log("G'aroyib son")
        }else{
            console.log("G'aroyib son emas")
        }
}
